## API 层规范

### 1. 接口定义规范

```typescript
import request from '@/utils/request'

/**
 * 用户查询参数
 */
export interface QueryUserParams {
  pageNum?: number
  pageSize?: number
  loginName?: string
  status?: number
}

/**
 * 用户信息
 */
export interface UserVO {
  userId: number
  loginName: string
  username: string
  phone?: string
  email?: string
  status: number
}

/**
 * 新增用户参数
 */
export interface AddUserParams {
  loginName: string
  username: string
  phone?: string
  email?: string
  roleIds?: number[]
}

/**
 * 修改用户参数 - 继承新增参数
 */
export interface ModifyUserParams extends AddUserParams {
  userId: number
}

/**
 * 分页结果
 */
export interface PageResult<T> {
  pageNum: number
  pageSize: number
  total: number
  pages: number
  rows: T[]
}

/**
 * 获取用户列表
 */
export const getUserList = (params: QueryUserParams): Promise<PageResult<UserVO>> => {
  return request.post('/user/getUserList', { body: params })
}

/**
 * 新增用户
 */
export const saveUser = (params: AddUserParams): Promise<void> => {
  return request.post('/user/saveUser', { body: params })
}

/**
 * 删除用户
 */
export const deleteUser = (params: { userId: number }): Promise<void> => {
  return request.post('/user/deleteUser', { body: params })
}
```

### 2. 命名约定

| 类型 | 命名规则 | 示例 |
|------|----------|------|
| 查询参数 | `Query` + 实体名 + `Params` | `QueryUserParams` |
| 新增参数 | `Add` + 实体名 + `Params` | `AddUserParams` |
| 修改参数 | `Modify` + 实体名 + `Params` | `ModifyUserParams` |
| 删除参数 | `Delete` + 实体名 + `Params` | `DeleteUserParams` |
| 响应实体 | 实体名 + `VO` / `Rsp` / `Info` | `UserVO`, `LoginRsp`, `ChannelInfo` |
| 分页结果 | `PageResult<T>` | `PageResult<UserVO>` |
| 查询方法 | `get` + 实体名 + `List` | `getUserList` |
| 详情方法 | `get` + 实体名 | `getUser` |
| 新增方法 | `save` + 实体名 | `saveUser` |
| 修改方法 | `modify` + 实体名 | `modifyUser` |
| 删除方法 | `delete` + 实体名 | `deleteUser` |

### 3. 请求规范

- **统一使用 POST 请求**，禁止使用 GET、PUT、DELETE
- 参数必须包裹在 `body` 中：`{ body: params }`
- 每个 API 函数必须有 JSDoc 注释
- 导出类型供其他模块使用

  searchForm.pageNum = 1

  searchForm.loginName = ''
  searchForm.status = undefined

    t('user.deleteConfirm'),
    t('common.confirm'),
    { type: 'warning' }
  ).then(async () => {
    await deleteUser({ userId: row.userId })
    ElMessage.success(t('common.success'))
    loadData()
  }).catch(() => {})
}

// 10. 生命周期
onMounted(() => {
  loadData()
})

// 11. Watch
watch(() => props.modelValue, (val) => {
  if (val) {
    // 弹窗打开时加载数据
  }

## Pinia Store 规范

### 1. Composition API 风格

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getUserInfo, type UserInfoRsp } from '@/api/auth'

/**
 * 用户状态管理Store
 */
export const useUserStore = defineStore('user', () => {
  // State
  const token = ref<string>('')
  const userInfo = ref<UserInfoRsp | null>(null)

  // Getters
  const isLoggedIn = computed(() => !!token.value)

  // Actions
  /**
   * 获取用户信息
   */
  const fetchUserInfo = async () => {
    const rsp = await getUserInfo()
    userInfo.value = rsp
    return rsp
  }

  /**
   * 用户登出
   */
  const logout = async () => {
    token.value = ''
    userInfo.value = null
    localStorage.removeItem('token')
  }

  return {
    token,
    userInfo,
    isLoggedIn,
    fetchUserInfo,
    logout
  }
})
```

## 路由规范

```typescript
import { createRouter, createWebHistory } from 'vue-router'
import type { RouteRecordRaw } from 'vue-router'

const routes: RouteRecordRaw[] = [
  {
    path: '/login',
    name: 'Login',
    component: () => import('@/views/login/index.vue'),
    meta: { public: true, title: 'login.title' }
  },
  {
    path: '/',
    component: () => import('@/layouts/MainLayout.vue'),
    redirect: '/dashboard',
    children: [
      {
        path: 'dashboard',
        name: 'Dashboard',
        component: () => import('@/views/dashboard/index.vue'),
        meta: { title: 'dashboard.title' }
      },
      {
        path: 'system',
        redirect: '/system/user',
        meta: { title: 'system.title' },
        children: [
          {
            path: 'user',
            name: 'SystemUser',
            component: () => import('@/views/system/user/index.vue'),
            meta: { title: 'system.user.title' }
          }
        ]
      }
    ]
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

// 路由守卫
router.beforeEach((to, _from, next) => {
  const token = localStorage.getItem('token')
  const isPublic = to.meta.public

  if (isPublic) {
    next()
    return
  }

  if (!token) {
    next({ path: '/login', query: { redirect: to.fullPath } })
    return
  }

  next()
})

export default router
```

  }

## 禁止事项

1. **禁止使用 GET/PUT/DELETE 请求** - 只能使用 POST
2. **禁止在代码中硬编码中文文本** - 必须使用 i18n
3. **禁止使用 `any` 类型** - 必须定义具体类型
4. **禁止在模板中使用复杂表达式** - 抽取为计算属性或方法
5. **禁止在循环中使用 `v-if`** - 使用计算属性过滤
6. **禁止直接修改 props** - 通过 emit 通知父组件
7. **禁止在 SCSS 中使用 `::v-deep`** - 使用 `:deep()`
8. **禁止混合使用 Options API 和 Composition API** - 统一使用 `<script setup>`
